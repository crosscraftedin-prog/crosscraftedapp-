   124→// URL sanitizer — blocks dangerous protocols (javascript:, data:, vbscript:,
   125→// file:) in markdown links + images. Only allows http(s), mailto, tel, and
   126→// relative URLs. Defense-in-depth on top of react-markdown's built-in filtering.
   127→function safeUrl(url: string | undefined): string | null {
   128→  if (!url || typeof url !== "string") return null;
   129→  const trimmed = url.trim();
   130→  if (!trimmed) return null;
   131→  if (trimmed.startsWith("/") || trimmed.startsWith("#")) return trimmed;
   132→  if (/^https?:\/\//i.test(trimmed) || /^mailto:/i.test(trimmed) || /^tel:/i.test(trimmed)) {
   133→    return trimmed;
   134→  }
   135→  return null;
   136→}
   137→
   138→function slugify(input: string): string {
   139→  return input
   140→    .toString()
   141→    .toLowerCase()
   142→    .trim()
   143→    .replace(/[^a-z0-9\s-]/g, "")
   144→    .replace(/[\s_-]+/g, "-")
   145→    .replace(/^-+|-+$/g, "")
   146→    .slice(0, 80);
   147→}
   148→
   149→// Parse a JSON array string safely — returns string[].
   150→function parseStringArray(raw: string | null): string[] {
   151→  if (!raw) return [];
   152→  try {
   153→    const parsed = JSON.parse(raw);
   154→    if (Array.isArray(parsed)) {
   155→      return parsed.map((v) => String(v ?? "")).filter((v) => v.length > 0);
   156→    }
   157→  } catch {
   158→    // ignore
   159→  }
   160→  return [];
   161→}
   162→
   163→function formatTimeAgo(iso: string): string {
   164→  const d = new Date(iso);
   165→  const diff = Date.now() - d.getTime();
   166→  const mins = Math.floor(diff / 60000);
   167→  if (mins < 60) return `${mins}m ago`;
   168→  const hrs = Math.floor(mins / 60);
   169→  if (hrs < 24) return `${hrs}h ago`;
   170→  const days = Math.floor(hrs / 24);
   171→  return `${days}d ago`;
   172→}

   697→            ))}
   698→          </div>
   699→        </div>
   700→      )}
   701→    </div>
   702→  );

   705→function ApologeticsEditorModal({
   706→  articleId,
   707→  onClose,
   708→  onSaved,
   709→  onDeleted,
   710→}: {
   711→  articleId: string | null;
   712→  onClose: () => void;
   713→  onSaved: () => void;
   714→  onDeleted: () => void;
   715→}) {
   716→  const [form, setForm] = useState<FormState>({ ...EMPTY_FORM });
   717→  const [loading, setLoading] = useState(articleId !== null);
   718→  const [saving, setSaving] = useState(false);
   719→  const [deleting, setDeleting] = useState(false);
   720→  const [confirmDelete, setConfirmDelete] = useState(false);
   721→  const [slugTouched, setSlugTouched] = useState(false);
   722→  const [currentSlug, setCurrentSlug] = useState<string>("");
   723→  const [currentStatus, setCurrentStatus] = useState<string>("draft");
   724→  // New state for the improved editor UX.
   725→  const [activeArticleId, setActiveArticleId] = useState<string | null>(articleId);
   726→  const [previewMode, setPreviewMode] = useState(false);
   727→  const [showDiscardConfirm, setShowDiscardConfirm] = useState(false);
   728→
   729→  // Refs: textarea for cursor-aware markdown insertion, snapshot for the
   730→  // unsaved-changes guard.
   731→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   732→  const snapshotRef = useRef<FormState>({ ...EMPTY_FORM });
   733→
   734→  // Deep-compare current form vs last-saved snapshot. JSON.stringify is fine
   735→  // here — the form is a small flat object.
   736→  const hasUnsavedChanges = useMemo(
   737→    () => JSON.stringify(form) !== JSON.stringify(snapshotRef.current),
   738→    [form]
   739→  );
   740→
   741→  // Load existing article if editing. For new articles, snapshot the empty
   742→  // form so hasUnsavedChanges starts false.
   743→  useEffect(() => {
   744→    if (!articleId) {
   745→      setLoading(false);
   746→      const initial = { ...EMPTY_FORM };
   747→      setForm(initial);
   748→      snapshotRef.current = initial;
   749→      return;
   750→    }
   751→    let cancelled = false;
   752→    (async () => {
   753→      try {
   754→        const res = await fetch(`/api/admin/articles/${articleId}`, { cache: "no-store" });
   755→        if (!res.ok) {
   756→          const data = await res.json().catch(() => ({}));
   757→          throw new Error(data?.error || `HTTP ${res.status}`);
   758→        }
   759→        const data = (await res.json()) as { article: KoinoArticle };
   760→        if (cancelled) return;
   761→        const fs = toFormState(data.article);
   762→        setForm(fs);
   763→        snapshotRef.current = fs;
   764→        setCurrentSlug(data.article.slug);
   765→        setCurrentStatus(data.article.status);
   766→        setActiveArticleId(data.article.id);
   767→      } catch (e: any) {
   768→        toast.error(e?.message || "Failed to load article");
   769→        onClose();
   770→      } finally {
   771→        if (!cancelled) setLoading(false);
   772→      }
   773→    })();
   774→    return () => {
   775→      cancelled = true;
   776→    };
   777→  }, [articleId, onClose]);
   778→
   779→  // Wrap onClose with the unsaved-changes guard. If the discard dialog is
   780→  // already open, Escape / overlay click just dismisses it instead of
   781→  // triggering another prompt.
   782→  const requestClose = useCallback(() => {
   783→    if (showDiscardConfirm) {
   784→      setShowDiscardConfirm(false);
   785→      return;
   786→    }
   787→    if (hasUnsavedChanges) {
   788→      setShowDiscardConfirm(true);
   789→    } else {
   790→      onClose();
   791→    }
   792→  }, [hasUnsavedChanges, showDiscardConfirm, onClose]);
   793→
   794→  // Escape to close (with guard) + lock body scroll while modal is open.
   795→  useEffect(() => {
   796→    const onKey = (e: KeyboardEvent) => {
   797→      if (e.key === "Escape") requestClose();
   798→    };
   799→    window.addEventListener("keydown", onKey);
   800→    document.body.style.overflow = "hidden";
   801→    return () => {
   802→      window.removeEventListener("keydown", onKey);
   803→      document.body.style.overflow = "";
   804→    };
   805→  }, [requestClose]);
   806→
   807→  const set = <K extends keyof FormState>(key: K, value: FormState[K]) => {
   808→    setForm((f) => ({ ...f, [key]: value }));
   809→  };
   810→
   811→  // Auto-generate slug from title if the admin hasn't manually touched the slug.
   812→  const effectiveSlug = slugTouched && form.slug ? form.slug : slugify(form.title);
   813→
   814→  // ─── Markdown toolbar ────────────────────────────────────────────────────
   815→  // Inserts markdown syntax at the cursor / around the selection. Updates
   816→  // form.content and restores focus + selection after React re-renders.
   817→  const applyInsert = (spec: InsertSpec) => {
   818→    const ta = textareaRef.current;
   819→    if (!ta) return;
   820→    const start = ta.selectionStart;
   821→    const end = ta.selectionEnd;
   822→    const value = ta.value;
   823→    const selected = value.slice(start, end);
   824→
   825→    let newValue: string;
   826→    let newSelStart: number;
   827→    let newSelEnd: number;
   828→
   829→    if (spec.mode === "snippet") {
   830→      newValue = value.slice(0, start) + spec.text + value.slice(end);
   831→      newSelStart = start + spec.text.length;
   832→      newSelEnd = newSelStart;
   833→    } else if (spec.mode === "block") {
   834→      // Insert the prefix at the start of the current line.
   835→      const lineStart = value.lastIndexOf("\n", start - 1) + 1;
   836→      const linePortion = value.slice(lineStart, start);
   837→      const inserted = spec.linePrefix + linePortion;
   838→      newValue = value.slice(0, lineStart) + inserted + value.slice(start);
   839→      newSelStart = start + spec.linePrefix.length;
   840→      newSelEnd = end + spec.linePrefix.length;
   841→    } else {
   842→      // Wrap selection (or insert placeholder) with prefix + suffix.
   843→      const placeholder = spec.placeholder ?? "text";
   844→      const inner = selected || placeholder;
   845→      const insertion = spec.prefix + inner + spec.suffix;
   846→      newValue = value.slice(0, start) + insertion + value.slice(end);
   847→      newSelStart = start + spec.prefix.length;
   848→      newSelEnd = newSelStart + inner.length;
   849→    }
   850→
   851→    set("content", newValue);
   852→    // Restore focus + selection after React re-renders the textarea with the
   853→    // new value.
   854→    requestAnimationFrame(() => {
   855→      const t = textareaRef.current;
   856→      if (!t) return;
   857→      t.focus();
   858→      try {
   859→        t.setSelectionRange(newSelStart, newSelEnd);
   860→      } catch {
   861→        // setSelectionRange can throw on some input types; safe to ignore.
   862→      }
   863→    });
   864→  };
   865→
   866→  // ─── Save (Draft / Publish / Unpublish) ─────────────────────────────────
   867→  // Keeps the editor OPEN after every successful save. Resets the unsaved-

   870→  const save = async (

   874→    if (!form.title.trim()) {

   876→      return;
   877→    }
   878→    if (!form.content.trim()) {
   879→      toast.error("Content is required");
   880→      return;
   881→    }
   882→    setSaving(true);
   883→    const wasPublished = currentStatus === "published";
   884→    try {
   885→      const payload: Record<string, unknown> = {
   886→        contentType: "APOLOGETICS",
   887→        title: form.title.trim(),
   888→        slug: form.slug.trim(), // server will slugify + de-dupe if empty
   889→        category: form.category.trim() || null,
   890→        subcategory: form.subcategory.trim() || null,
   891→        difficulty: form.difficulty || null,
   892→        excerpt: form.excerpt,
   893→        shortAnswer: form.shortAnswer,
   894→        content: form.content,
   895→        coverImageUrl: form.coverImageUrl,
   896→        authorName: form.authorName,
   897→        bibleRefs: form.bibleRefs, // CSV → server coerces to JSON array
   898→        status: targetStatus, // "draft" | "published"
   899→        featured: form.featured,
   900→        seoTitle: form.seoTitle,
   901→        seoDescription: form.seoDescription,
   902→        canonicalUrl: form.canonicalUrl,
   903→      };
   904→      const idForSave = activeArticleId;
   905→      const url = idForSave ? `/api/admin/articles/${idForSave}` : "/api/admin/articles";
   906→      const method = idForSave ? "PATCH" : "POST";
   907→      const res = await fetch(url, {
   908→        method,
   909→        headers: { "Content-Type": "application/json" },
   910→        body: JSON.stringify(payload),
   911→      });
   912→      if (!res.ok) {
   913→        const data = await res.json().catch(() => ({}));
   914→        throw new Error(data?.error || `HTTP ${res.status}`);
   915→      }
   916→      const data = (await res.json()) as { article: KoinoArticle };
   917→      const updated = toFormState(data.article);
   918→      // Sync form + reset snapshot so the close guard no longer fires.
   919→      setForm(updated);
   920→      snapshotRef.current = updated;
   921→      setActiveArticleId(data.article.id);
   922→      setCurrentSlug(data.article.slug);
   923→      setCurrentStatus(data.article.status);
   924→      setConfirmDelete(false);
   925→

   928→        toast.success(opts.successToast);
   929→      } else if (opts?.wasUnpublish) {
   930→        toast.success("Article unpublished (reverted to draft)");
   931→      } else if (targetStatus === "published") {
   932→        toast.success(wasPublished ? "Article updated" : "Article published");
   933→      } else {
   934→        toast.success("Draft saved");
   935→      }
   936→
   937→      // Refresh the parent list (does NOT close the editor).
   938→      onSaved();
   939→    } catch (e: any) {
   940→      toast.error(e?.message || "Failed to save article");
   941→      // Do NOT update the snapshot — leave hasUnsavedChanges true so the
   942→      // close guard still fires.
   943→    } finally {
   944→      setSaving(false);
   945→    }
   946→  };

   958→      }

   960→      onDeleted();
   961→    } catch (e: any) {

   963→    } finally {
   964→      setDeleting(false);
   965→    }
   966→  };
   967→
   968→  if (loading) {
   969→    return (
   970→      <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/60 backdrop-blur-sm">
   971→        <div className="w-8 h-8 rounded-full border-2 border-transparent border-t-[#7C3AED] animate-spin" />
   972→      </div>
   973→    );
   974→  }
   975→
   976→  return (
   977→    <div

   980→    >
   981→      <div

   983→        onClick={(e) => e.stopPropagation()}

   985→        {/* Header */}
   986→        <div className="flex items-center justify-between mb-4">
   987→          <div className="flex items-center gap-2 min-w-0">
   988→            <FileText size={14} className="text-[#A78BFA] shrink-0" />

   990→              {activeArticleId ? "Edit Article" : "New Article"}
   991→            </h2>
   992→            {hasUnsavedChanges && (
   993→              <span className="shrink-0 text-[9px] font-bold uppercase tracking-wider px-1.5 py-0.5 rounded-md bg-[#F59E0B]/15 text-[#F59E0B]">
   994→                Unsaved
   995→              </span>
   996→            )}
   997→          </div>
   998→          <button
   999→            onClick={requestClose}
  1000→            className="shrink-0 w-7 h-7 rounded-lg bg-white/[0.04] hover:bg-white/[0.08] flex items-center justify-center text-[#94A3B8] hover:text-white"
  1001→            aria-label="Close"
  1002→          >
  1003→            <X size={14} />
  1004→          </button>
  1005→        </div>
  1006→
  1007→        {/* Body — collapsible sections */}
  1008→        <div className="space-y-3">
  1009→          {/* ▸ ARTICLE INFORMATION */}
  1010→          <Section title="Article Information">
  1011→            <Field label="Title" required>
  1012→              <input
  1013→                type="text"
  1014→                value={form.title}
  1015→                onChange={(e) => set("title", e.target.value)}
  1016→                className="neo-input text-sm"
  1017→                placeholder="Did Jesus Really Rise from the Dead?"
  1018→              />
  1019→            </Field>
  1020→
  1021→            <Field label="Slug" hint="URL-friendly. Auto-generated from title if empty.">
