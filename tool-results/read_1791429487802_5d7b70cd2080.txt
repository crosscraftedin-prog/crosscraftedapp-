   391→function MembersTab() {
   392→  const [members, setMembers] = useState<any[]>([]);
   393→  const [stats, setStats] = useState<any>(null);
   394→  const [loading, setLoading] = useState(true);
   395→  const [search, setSearch] = useState("");
   396→  const [filterState, setFilterState] = useState("");
   397→  const [filterStatus, setFilterStatus] = useState("");
   398→  const [filterVerified, setFilterVerified] = useState("");
   399→  const [total, setTotal] = useState(0);
   400→  const [page, setPage] = useState(0);
   401→  const limit = 20;
   402→  const [selectedMember, setSelectedMember] = useState<any>(null);
   403→  const [memberDetail, setMemberDetail] = useState<any>(null);
   404→  const [detailLoading, setDetailLoading] = useState(false);
   405→  const [saving, setSaving] = useState(false);
   406→
   407→  const loadMembers = useCallback(async () => {
   408→    setLoading(true);
   409→    try {
   410→      const params = new URLSearchParams({ limit: String(limit), offset: String(page * limit) });
   411→      if (search) params.set("search", search);
   412→      if (filterState) params.set("state", filterState);
   413→      if (filterStatus) params.set("profileCompleted", filterStatus);
   414→      const res = await fetch(`/api/admin/members?${params}`);
   415→      const data = await res.json();
   416→      setMembers(data.members || []);
   417→      setTotal(data.total || 0);
   418→    } catch { setMembers([]); }
   419→    finally { setLoading(false); }
   420→  }, [search, filterState, filterStatus, page]);
   421→
   422→  const loadStats = useCallback(async () => {
   423→    try {
   424→      const res = await fetch("/api/admin/members?stats=true");
   425→      const data = await res.json();
   426→      setStats(data);
   427→    } catch {}
   428→  }, []);
   429→
   430→  useEffect(() => { loadMembers(); loadStats(); }, [loadMembers, loadStats]);
   431→
   432→  const openMember = async (id: string) => {
   433→    setSelectedMember(id);
   434→    setDetailLoading(true);
   435→    try {
   436→      const res = await fetch(`/api/admin/members/${id}`);
   437→      const data = await res.json();
   438→      setMemberDetail(data.member);
   439→    } catch {}
   440→    finally { setDetailLoading(false); }
   441→  };
   442→
   443→  const updateMember = async (updates: any) => {
   444→    if (!memberDetail) return;
   445→    setSaving(true);
   446→    try {
   447→      const res = await fetch(`/api/admin/members/${memberDetail.id}`, {
   448→        method: "PATCH",
   449→        headers: { "Content-Type": "application/json" },
   450→        body: JSON.stringify(updates),
   451→      });
   452→      const data = await res.json();
   453→      if (!res.ok) throw new Error(data.error);
   454→      setMemberDetail(data.member);
   455→      toast.success("Member updated");

   701→function ChurchesTab() {

   704→  const [statusFilter, setStatusFilter] = useState<"all" | "PENDING" | "PUBLISHED" | "REJECTED" | "CANCELLED">("all");

   712→    setLoading(true);

   715→      const url = status === "all" ? "/api/admin/churches" : `/api/admin/churches?status=${status}`;

   719→        const apiError = data?.error || `Failed to load churches (HTTP ${res.status})`;
   720→        const prismaCode = data?.prismaCode || data?.code;
   721→        const detail = data?.detail;
   722→        const fullError = prismaCode
   723→          ? `${apiError} (Prisma code: ${prismaCode}${detail ? ` — ${detail}` : ""})`
   724→          : apiError;
   725→        throw new Error(fullError);
