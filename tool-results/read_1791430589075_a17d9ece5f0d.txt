   124→   124→// URL sanitizer — blocks dangerous protocols (javascript:, data:, vbscript:,
   125→   125→// file:) in markdown links + images. Only allows http(s), mailto, tel, and
   126→   126→// relative URLs. Defense-in-depth on top of react-markdown's built-in filtering.
   127→   127→function safeUrl(url: string | undefined): string | null {
   128→   128→  if (!url || typeof url !== "string") return null;
   129→   129→  const trimmed = url.trim();
   130→   130→  if (!trimmed) return null;
   131→   131→  if (trimmed.startsWith("/") || trimmed.startsWith("#")) return trimmed;
   132→   132→  if (/^https?:\/\//i.test(trimmed) || /^mailto:/i.test(trimmed) || /^tel:/i.test(trimmed)) {
   133→   133→    return trimmed;
   134→   134→  }
   135→   135→  return null;
   136→   136→}
   137→   137→
   138→   138→function slugify(input: string): string {
   139→   139→  return input
   140→   140→    .toString()
   141→   141→    .toLowerCase()
   142→   142→    .trim()
   143→   143→    .replace(/[^a-z0-9\s-]/g, "")
   144→   144→    .replace(/[\s_-]+/g, "-")
   145→   145→    .replace(/^-+|-+$/g, "")
   146→   146→    .slice(0, 80);
   147→   147→}
   148→   148→
   149→   149→// Parse a JSON array string safely — returns string[].
   150→   150→function parseStringArray(raw: string | null): string[] {
   151→   151→  if (!raw) return [];
   152→   152→  try {
   153→   153→    const parsed = JSON.parse(raw);
   154→   154→    if (Array.isArray(parsed)) {
   155→   155→      return parsed.map((v) => String(v ?? "")).filter((v) => v.length > 0);
   156→   156→    }
   157→   157→  } catch {
   158→   158→    // ignore
   159→   159→  }
   160→   160→  return [];
   161→   161→}
   162→   162→
   163→   163→function formatTimeAgo(iso: string): string {
   164→   164→  const d = new Date(iso);
   165→   165→  const diff = Date.now() - d.getTime();
   166→   166→  const mins = Math.floor(diff / 60000);
   167→   167→  if (mins < 60) return `${mins}m ago`;
   168→   168→  const hrs = Math.floor(mins / 60);
   169→   169→  if (hrs < 24) return `${hrs}h ago`;
   170→   170→  const days = Math.floor(hrs / 24);
   171→   171→  return `${days}d ago`;
   172→   172→}

   705→   705→function ApologeticsEditorModal({
   706→   706→  articleId,
   707→   707→  onClose,
   708→   708→  onSaved,
   709→   709→  onDeleted,
   710→   710→}: {
   711→   711→  articleId: string | null;
   712→   712→  onClose: () => void;
   713→   713→  onSaved: () => void;
   714→   714→  onDeleted: () => void;
   715→   715→}) {
   716→   716→  const [form, setForm] = useState<FormState>({ ...EMPTY_FORM });
   717→   717→  const [loading, setLoading] = useState(articleId !== null);
   718→   718→  const [saving, setSaving] = useState(false);
   719→   719→  const [deleting, setDeleting] = useState(false);
   720→   720→  const [confirmDelete, setConfirmDelete] = useState(false);
   721→   721→  const [slugTouched, setSlugTouched] = useState(false);
   722→   722→  const [currentSlug, setCurrentSlug] = useState<string>("");
   723→   723→  const [currentStatus, setCurrentStatus] = useState<string>("draft");
   724→   724→  // New state for the improved editor UX.
   725→   725→  const [activeArticleId, setActiveArticleId] = useState<string | null>(articleId);
   726→   726→  const [previewMode, setPreviewMode] = useState(false);
   727→   727→  const [showDiscardConfirm, setShowDiscardConfirm] = useState(false);
   728→   728→
   729→   729→  // Refs: textarea for cursor-aware markdown insertion, snapshot for the
   730→   730→  // unsaved-changes guard.
   731→   731→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   732→   732→  const snapshotRef = useRef<FormState>({ ...EMPTY_FORM });
   733→   733→
   734→   734→  // Deep-compare current form vs last-saved snapshot. JSON.stringify is fine
   735→   735→  // here — the form is a small flat object.
   736→   736→  const hasUnsavedChanges = useMemo(
   737→   737→    () => JSON.stringify(form) !== JSON.stringify(snapshotRef.current),
   738→   738→    [form]
   739→   739→  );
   740→   740→
   741→   741→  // Load existing article if editing. For new articles, snapshot the empty
   742→   742→  // form so hasUnsavedChanges starts false.
   743→   743→  useEffect(() => {
   744→   744→    if (!articleId) {
   745→   745→      setLoading(false);
   746→   746→      const initial = { ...EMPTY_FORM };
   747→   747→      setForm(initial);
   748→   748→      snapshotRef.current = initial;
   749→   749→      return;
   750→   750→    }
   751→   751→    let cancelled = false;
   752→   752→    (async () => {
   753→   753→      try {
   754→   754→        const res = await fetch(`/api/admin/articles/${articleId}`, { cache: "no-store" });
   755→   755→        if (!res.ok) {
   756→   756→          const data = await res.json().catch(() => ({}));
   757→   757→          throw new Error(data?.error || `HTTP ${res.status}`);
   758→   758→        }
   759→   759→        const data = (await res.json()) as { article: KoinoArticle };
   760→   760→        if (cancelled) return;
   761→   761→        const fs = toFormState(data.article);
   762→   762→        setForm(fs);
   763→   763→        snapshotRef.current = fs;
   764→   764→        setCurrentSlug(data.article.slug);
   765→   765→        setCurrentStatus(data.article.status);
   766→   766→        setActiveArticleId(data.article.id);
   767→   767→      } catch (e: any) {
   768→   768→        toast.error(e?.message || "Failed to load article");
   769→   769→        onClose();
   770→   770→      } finally {
   771→   771→        if (!cancelled) setLoading(false);
   772→   772→      }
   773→   773→    })();
   774→   774→    return () => {
   775→   775→      cancelled = true;
   776→   776→    };
   777→   777→  }, [articleId, onClose]);
   778→   778→
   779→   779→  // Wrap onClose with the unsaved-changes guard. If the discard dialog is
   780→   780→  // already open, Escape / overlay click just dismisses it instead of
   781→   781→  // triggering another prompt.
   782→   782→  const requestClose = useCallback(() => {
   783→   783→    if (showDiscardConfirm) {
   784→   784→      setShowDiscardConfirm(false);
   785→   785→      return;
   786→   786→    }
   787→   787→    if (hasUnsavedChanges) {
   788→   788→      setShowDiscardConfirm(true);
   789→   789→    } else {
   790→   790→      onClose();
   791→   791→    }
   792→   792→  }, [hasUnsavedChanges, showDiscardConfirm, onClose]);
   793→   793→
   794→   794→  // Escape to close (with guard) + lock body scroll while modal is open.
   795→   795→  useEffect(() => {
   796→   796→    const onKey = (e: KeyboardEvent) => {
   797→   797→      if (e.key === "Escape") requestClose();
   798→   798→    };
   799→   799→    window.addEventListener("keydown", onKey);
   800→   800→    document.body.style.overflow = "hidden";
   801→   801→    return () => {
   802→   802→      window.removeEventListener("keydown", onKey);
   803→   803→      document.body.style.overflow = "";
   804→   804→    };
   805→   805→  }, [requestClose]);
   806→   806→
   807→   807→  const set = <K extends keyof FormState>(key: K, value: FormState[K]) => {
   808→   808→    setForm((f) => ({ ...f, [key]: value }));
   809→   809→  };
   810→   810→
   811→   811→  // Auto-generate slug from title if the admin hasn't manually touched the slug.
   812→   812→  const effectiveSlug = slugTouched && form.slug ? form.slug : slugify(form.title);
   813→   813→
   814→   814→  // ─── Markdown toolbar ────────────────────────────────────────────────────
   815→   815→  // Inserts markdown syntax at the cursor / around the selection. Updates
   816→   816→  // form.content and restores focus + selection after React re-renders.
   817→   817→  const applyInsert = (spec: InsertSpec) => {
   818→   818→    const ta = textareaRef.current;
   819→   819→    if (!ta) return;
   820→   820→    const start = ta.selectionStart;
   821→   821→    const end = ta.selectionEnd;
   822→   822→    const value = ta.value;
   823→   823→    const selected = value.slice(start, end);
   824→   824→
   825→   825→    let newValue: string;
   826→   826→    let newSelStart: number;
   827→   827→    let newSelEnd: number;
   828→   828→
   829→   829→    if (spec.mode === "snippet") {
   830→   830→      newValue = value.slice(0, start) + spec.text + value.slice(end);
   831→   831→      newSelStart = start + spec.text.length;
   832→   832→      newSelEnd = newSelStart;
   833→   833→    } else if (spec.mode === "block") {
   834→   834→      // Insert the prefix at the start of the current line.
   835→   835→      const lineStart = value.lastIndexOf("\n", start - 1) + 1;
   836→   836→      const linePortion = value.slice(lineStart, start);
   837→   837→      const inserted = spec.linePrefix + linePortion;
   838→   838→      newValue = value.slice(0, lineStart) + inserted + value.slice(start);
   839→   839→      newSelStart = start + spec.linePrefix.length;
   840→   840→      newSelEnd = end + spec.linePrefix.length;
   841→   841→    } else {
   842→   842→      // Wrap selection (or insert placeholder) with prefix + suffix.
   843→   843→      const placeholder = spec.placeholder ?? "text";
   844→   844→      const inner = selected || placeholder;
   845→   845→      const insertion = spec.prefix + inner + spec.suffix;
   846→   846→      newValue = value.slice(0, start) + insertion + value.slice(end);
   847→   847→      newSelStart = start + spec.prefix.length;
   848→   848→      newSelEnd = newSelStart + inner.length;
   849→   849→    }
   850→   850→
   851→   851→    set("content", newValue);
   852→   852→    // Restore focus + selection after React re-renders the textarea with the
   853→   853→    // new value.
   854→   854→    requestAnimationFrame(() => {
   855→   855→      const t = textareaRef.current;
   856→   856→      if (!t) return;
   857→   857→      t.focus();
   858→   858→      try {
   859→   859→        t.setSelectionRange(newSelStart, newSelEnd);
   860→   860→      } catch {
   861→   861→        // setSelectionRange can throw on some input types; safe to ignore.
   862→   862→      }
   863→   863→    });
   864→   864→  };
   865→   865→
   866→   866→  // ─── Save (Draft / Publish / Unpublish) ─────────────────────────────────
   867→   867→  // Keeps the editor OPEN after every successful save. Resets the unsaved-
   868→   868→  // changes snapshot so the close guard doesn't fire. On failure, leaves the
   869→   869→  // snapshot alone so the guard still works.
   870→   870→  const save = async (
   871→   871→    targetStatus: "draft" | "published",
   872→   872→    opts?: { successToast?: string; wasUnpublish?: boolean }
   873→   873→  ) => {
   874→   874→    if (!form.title.trim()) {
   875→   875→      toast.error("Title is required");
   876→   876→      return;
   877→   877→    }
   878→   878→    if (!form.content.trim()) {
   879→   879→      toast.error("Content is required");
   880→   880→      return;
   881→   881→    }
   882→   882→    setSaving(true);
   883→   883→    const wasPublished = currentStatus === "published";
   884→   884→    try {
   885→   885→      const payload: Record<string, unknown> = {
   886→   886→        contentType: "APOLOGETICS",
   887→   887→        title: form.title.trim(),
   888→   888→        slug: form.slug.trim(), // server will slugify + de-dupe if empty
   889→   889→        category: form.category.trim() || null,
   890→   890→        subcategory: form.subcategory.trim() || null,
   891→   891→        difficulty: form.difficulty || null,
   892→   892→        excerpt: form.excerpt,
   893→   893→        shortAnswer: form.shortAnswer,
   894→   894→        content: form.content,
   895→   895→        coverImageUrl: form.coverImageUrl,
   896→   896→        authorName: form.authorName,
   897→   897→        bibleRefs: form.bibleRefs, // CSV → server coerces to JSON array
   898→   898→        status: targetStatus, // "draft" | "published"
   899→   899→        featured: form.featured,
   900→   900→        seoTitle: form.seoTitle,
   901→   901→        seoDescription: form.seoDescription,
   902→   902→        canonicalUrl: form.canonicalUrl,
   903→   903→      };
   904→   904→      const idForSave = activeArticleId;
   905→   905→      const url = idForSave ? `/api/admin/articles/${idForSave}` : "/api/admin/articles";
   906→   906→      const method = idForSave ? "PATCH" : "POST";
   907→   907→      const res = await fetch(url, {
   908→   908→        method,
   909→   909→        headers: { "Content-Type": "application/json" },
   910→   910→        body: JSON.stringify(payload),
   911→   911→      });
   912→   912→      if (!res.ok) {
   913→   913→        const data = await res.json().catch(() => ({}));
   914→   914→        throw new Error(data?.error || `HTTP ${res.status}`);
   915→   915→      }
   916→   916→      const data = (await res.json()) as { article: KoinoArticle };
   917→   917→      const updated = toFormState(data.article);
   918→   918→      // Sync form + reset snapshot so the close guard no longer fires.
   919→   919→      setForm(updated);
   920→   920→      snapshotRef.current = updated;
   921→   921→      setActiveArticleId(data.article.id);
   922→   922→      setCurrentSlug(data.article.slug);
   923→   923→      setCurrentStatus(data.article.status);
