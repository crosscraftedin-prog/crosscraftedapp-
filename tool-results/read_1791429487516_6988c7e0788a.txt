    31→// ─── Helper: map admin category label → APOLOGETICS_TOPICS id ──────────────
    32→// The admin editor uses category labels like "God & Existence", "Jesus Christ",
    33→// "Resurrection", "Bible", "Science & Faith", etc. The SPA's topic filter uses
    34→// IDs like "gods_existence", "resurrection", "bible_reliability".
    35→// This helper maps the label to the closest topic ID so the filter works.
    36→// If no match, returns "" (shows under "All Topics").
    37→function mapCategoryToTopicId(category: string | null | undefined): string {
    38→  if (!category) return "";
    39→  const c = category.toLowerCase().trim();
    40→  if (c.includes("god") && c.includes("exist")) return "gods_existence";
    41→  if (c.includes("evil") || c.includes("suffer")) return "problem_of_evil";
    42→  if (c.includes("resurrection") || c.includes("jesus")) return "resurrection";
    43→  if (c.includes("bible")) return "bible_reliability";
    44→  if (c.includes("science")) return "science_faith";
    45→  if (c.includes("world") || c.includes("religion")) return "world_religions";
    46→  return ""; // no topic match — shows under "All Topics"
    47→}
