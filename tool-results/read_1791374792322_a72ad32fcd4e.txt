   367→function MembersTab() {
   368→  const [members, setMembers] = useState<any[]>([]);
   369→  const [stats, setStats] = useState<any>(null);
   370→  const [loading, setLoading] = useState(true);
   371→  const [search, setSearch] = useState("");
   372→  const [filterState, setFilterState] = useState("");
   373→  const [filterStatus, setFilterStatus] = useState("");
   374→  const [filterVerified, setFilterVerified] = useState("");
   375→  const [total, setTotal] = useState(0);
   376→  const [page, setPage] = useState(0);
   377→  const limit = 20;
   378→  const [selectedMember, setSelectedMember] = useState<any>(null);
   379→  const [memberDetail, setMemberDetail] = useState<any>(null);
   380→  const [detailLoading, setDetailLoading] = useState(false);
   381→  const [saving, setSaving] = useState(false);
   382→
   383→  const loadMembers = useCallback(async () => {
   384→    setLoading(true);
   385→    try {
   386→      const params = new URLSearchParams({ limit: String(limit), offset: String(page * limit) });
   387→      if (search) params.set("search", search);
   388→      if (filterState) params.set("state", filterState);
   389→      if (filterStatus) params.set("profileCompleted", filterStatus);
   390→      const res = await fetch(`/api/admin/members?${params}`);
   391→      const data = await res.json();
   392→      setMembers(data.members || []);
   393→      setTotal(data.total || 0);
   394→    } catch { setMembers([]); }
   395→    finally { setLoading(false); }
   396→  }, [search, filterState, filterStatus, page]);
   397→
   398→  const loadStats = useCallback(async () => {
   399→    try {
   400→      const res = await fetch("/api/admin/members?stats=true");
   401→      const data = await res.json();
   402→      setStats(data);
   403→    } catch {}
   404→  }, []);
   405→
   406→  useEffect(() => { loadMembers(); loadStats(); }, [loadMembers, loadStats]);
   407→
   408→  const openMember = async (id: string) => {
   409→    setSelectedMember(id);
   410→    setDetailLoading(true);
   411→    try {
   412→      const res = await fetch(`/api/admin/members/${id}`);
   413→      const data = await res.json();
   414→      setMemberDetail(data.member);
   415→    } catch {}
   416→    finally { setDetailLoading(false); }
   417→  };
   418→
   419→  const updateMember = async (updates: any) => {
   420→    if (!memberDetail) return;
   421→    setSaving(true);
   422→    try {
   423→      const res = await fetch(`/api/admin/members/${memberDetail.id}`, {
   424→        method: "PATCH",
   425→        headers: { "Content-Type": "application/json" },
   426→        body: JSON.stringify(updates),
   427→      });
   428→      const data = await res.json();
   429→      if (!res.ok) throw new Error(data.error);
   430→      setMemberDetail(data.member);
   431→      toast.success("Member updated");
   432→      loadMembers();
   433→    } catch (e: any) {
   434→      toast.error(e.message || "Failed to update");
   435→    } finally { setSaving(false); }
   436→  };
   437→
   438→  const togglePermission = (perm: string) => {
   439→    if (!memberDetail) return;
   440→    const current: string[] = JSON.parse(memberDetail.permissions || "[]");
   441→    const next = current.includes(perm) ? current.filter(p => p !== perm) : [...current, perm];
   442→    updateMember({ permissions: next });
   443→  };
   444→
   445→  return (
   446→    <div className="space-y-4">
   447→      <PreviewModeBanner section="Members" />
   448→
   449→      {/* Stats */}
   450→      {stats && (
   451→        <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   452→          <div className="bg-[#1C1929] border border-white/[0.06] rounded-xl p-3">
   453→            <p className="text-xl font-extrabold text-white">{stats.totalMembers}</p>
   454→            <p className="text-[9px] text-[#94A3B8] uppercase">Total Members</p>
   455→          </div>
   456→          <div className="bg-[#1C1929] border border-white/[0.06] rounded-xl p-3">
   457→            <p className="text-xl font-extrabold text-[#22C55E]">{stats.newToday}</p>
