   386→function MembersTab() {
   387→  const [members, setMembers] = useState<any[]>([]);
   388→  const [stats, setStats] = useState<any>(null);
   389→  const [loading, setLoading] = useState(true);
   390→  const [search, setSearch] = useState("");
   391→  const [filterState, setFilterState] = useState("");
   392→  const [filterStatus, setFilterStatus] = useState("");
   393→  const [filterVerified, setFilterVerified] = useState("");
   394→  const [total, setTotal] = useState(0);
   395→  const [page, setPage] = useState(0);
   396→  const limit = 20;
   397→  const [selectedMember, setSelectedMember] = useState<any>(null);
   398→  const [memberDetail, setMemberDetail] = useState<any>(null);
   399→  const [detailLoading, setDetailLoading] = useState(false);
   400→  const [saving, setSaving] = useState(false);
   401→
   402→  const loadMembers = useCallback(async () => {
   403→    setLoading(true);
   404→    try {
   405→      const params = new URLSearchParams({ limit: String(limit), offset: String(page * limit) });
   406→      if (search) params.set("search", search);
   407→      if (filterState) params.set("state", filterState);
   408→      if (filterStatus) params.set("profileCompleted", filterStatus);
   409→      const res = await fetch(`/api/admin/members?${params}`);
   410→      const data = await res.json();
   411→      setMembers(data.members || []);
   412→      setTotal(data.total || 0);
   413→    } catch { setMembers([]); }
   414→    finally { setLoading(false); }
   415→  }, [search, filterState, filterStatus, page]);
   416→
   417→  const loadStats = useCallback(async () => {
   418→    try {
   419→      const res = await fetch("/api/admin/members?stats=true");
   420→      const data = await res.json();
   421→      setStats(data);
   422→    } catch {}
   423→  }, []);
   424→
   425→  useEffect(() => { loadMembers(); loadStats(); }, [loadMembers, loadStats]);
   426→
   427→  const openMember = async (id: string) => {
   428→    setSelectedMember(id);
   429→    setDetailLoading(true);
   430→    try {
   431→      const res = await fetch(`/api/admin/members/${id}`);
   432→      const data = await res.json();
   433→      setMemberDetail(data.member);
   434→    } catch {}
   435→    finally { setDetailLoading(false); }
   436→  };
   437→
   438→  const updateMember = async (updates: any) => {
   439→    if (!memberDetail) return;
   440→    setSaving(true);
   441→    try {
   442→      const res = await fetch(`/api/admin/members/${memberDetail.id}`, {
   443→        method: "PATCH",
   444→        headers: { "Content-Type": "application/json" },
   445→        body: JSON.stringify(updates),
   446→      });
   447→      const data = await res.json();
   448→      if (!res.ok) throw new Error(data.error);
   449→      setMemberDetail(data.member);
   450→      toast.success("Member updated");
   451→      loadMembers();
   452→    } catch (e: any) {
   453→      toast.error(e.message || "Failed to update");
   454→    } finally { setSaving(false); }
   455→  };
   456→
