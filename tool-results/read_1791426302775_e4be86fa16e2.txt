   110→   110→function slugify(input: string): string {
   111→   111→  return input
   112→   112→    .toString()
   113→   113→    .toLowerCase()
   114→   114→    .trim()
   115→   115→    .replace(/[^a-z0-9\s-]/g, "")
   116→   116→    .replace(/[\s_-]+/g, "-")
   117→   117→    .replace(/^-+|-+$/g, "")
   118→   118→    .slice(0, 80);
   119→   119→}
   120→   120→
   121→   121→// Parse a JSON array string safely — returns string[].
   122→   122→function parseStringArray(raw: string | null): string[] {
   123→   123→  if (!raw) return [];
   124→   124→  try {
   125→   125→    const parsed = JSON.parse(raw);
   126→   126→    if (Array.isArray(parsed)) {
   127→   127→      return parsed.map((v) => String(v ?? "")).filter((v) => v.length > 0);
   128→   128→    }
   129→   129→  } catch {
   130→   130→    // ignore
   131→   131→  }
   132→   132→  return [];
   133→   133→}
   134→   134→
   135→   135→function formatTimeAgo(iso: string): string {
   136→   136→  const d = new Date(iso);
   137→   137→  const diff = Date.now() - d.getTime();
   138→   138→  const mins = Math.floor(diff / 60000);
   139→   139→  if (mins < 60) return `${mins}m ago`;
   140→   140→  const hrs = Math.floor(mins / 60);
   141→   141→  if (hrs < 24) return `${hrs}h ago`;
   142→   142→  const days = Math.floor(hrs / 24);
   143→   143→  return `${days}d ago`;
   144→   144→}

   441→   441→function ApologeticsEditorModal({
