   110→function slugify(input: string): string {
   111→  return input
   112→    .toString()
   113→    .toLowerCase()
   114→    .trim()
   115→    .replace(/[^a-z0-9\s-]/g, "")
   116→    .replace(/[\s_-]+/g, "-")
   117→    .replace(/^-+|-+$/g, "")
   118→    .slice(0, 80);
   119→}
   120→
   121→// Parse a JSON array string safely — returns string[].
   122→function parseStringArray(raw: string | null): string[] {
   123→  if (!raw) return [];
   124→  try {
   125→    const parsed = JSON.parse(raw);
   126→    if (Array.isArray(parsed)) {
   127→      return parsed.map((v) => String(v ?? "")).filter((v) => v.length > 0);
   128→    }
   129→  } catch {
   130→    // ignore
   131→  }
   132→  return [];
   133→}
   134→
   135→function formatTimeAgo(iso: string): string {
   136→  const d = new Date(iso);
   137→  const diff = Date.now() - d.getTime();
   138→  const mins = Math.floor(diff / 60000);
   139→  if (mins < 60) return `${mins}m ago`;
   140→  const hrs = Math.floor(mins / 60);
   141→  if (hrs < 24) return `${hrs}h ago`;
   142→  const days = Math.floor(hrs / 24);
   143→  return `${days}d ago`;
   144→}

   153→    difficulty: article.difficulty ?? "",

   441→function ApologeticsEditorModal({
   442→  articleId,
   443→  onClose,
   444→  onSaved,
   445→  onDeleted,
   446→}: {
   447→  articleId: string | null;
   448→  onClose: () => void;
   449→  onSaved: () => void;
   450→  onDeleted: () => void;
   451→}) {
   452→  const [form, setForm] = useState<FormState>({ ...EMPTY_FORM });
   453→  const [loading, setLoading] = useState(articleId !== null);
   454→  const [saving, setSaving] = useState(false);
   455→  const [deleting, setDeleting] = useState(false);
   456→  const [confirmDelete, setConfirmDelete] = useState(false);
   457→  const [slugTouched, setSlugTouched] = useState(false);
   458→  const [currentSlug, setCurrentSlug] = useState<string>("");
   459→  const [currentStatus, setCurrentStatus] = useState<string>("draft");
   460→
   461→  // Load existing article if editing.
   462→  useEffect(() => {
   463→    if (!articleId) {
   464→      setLoading(false);
   465→      return;
   466→    }
   467→    let cancelled = false;
   468→    (async () => {
   469→      try {
   470→        const res = await fetch(`/api/admin/articles/${articleId}`, { cache: "no-store" });
   471→        if (!res.ok) {
   472→          const data = await res.json().catch(() => ({}));
   473→          throw new Error(data?.error || `HTTP ${res.status}`);
   474→        }
   475→        const data = (await res.json()) as { article: KoinoArticle };
   476→        if (cancelled) return;
   477→        setForm(toFormState(data.article));
   478→        setCurrentSlug(data.article.slug);
   479→        setCurrentStatus(data.article.status);
   480→      } catch (e: any) {
   481→        toast.error(e?.message || "Failed to load article");
   482→        onClose();
   483→      } finally {
   484→        if (!cancelled) setLoading(false);
   485→      }
   486→    })();
   487→    return () => {
   488→      cancelled = true;
   489→    };
   490→  }, [articleId, onClose]);
   491→
   492→  // Close on Escape + lock scroll.
   493→  useEffect(() => {
   494→    const onKey = (e: KeyboardEvent) => {
   495→      if (e.key === "Escape") onClose();
   496→    };
   497→    window.addEventListener("keydown", onKey);
   498→    document.body.style.overflow = "hidden";
   499→    return () => {
   500→      window.removeEventListener("keydown", onKey);
   501→      document.body.style.overflow = "";
   502→    };
   503→  }, [onClose]);
   504→
   505→  const set = <K extends keyof FormState>(key: K, value: FormState[K]) => {
   506→    setForm((f) => ({ ...f, [key]: value }));
   507→  };
   508→
   509→  // Auto-generate slug from title if the admin hasn't manually touched the slug.
   510→  const effectiveSlug = slugTouched && form.slug ? form.slug : slugify(form.title);
   511→
   512→  const save = async () => {
   513→    if (!form.title.trim()) {
   514→      toast.error("Title is required");
   515→      return;
   516→    }
   517→    if (!form.content.trim()) {
   518→      toast.error("Content is required");
   519→      return;
   520→    }
   521→    setSaving(true);
   522→    try {
   523→      const payload: Record<string, unknown> = {
   524→        contentType: "APOLOGETICS",
   525→        title: form.title.trim(),
   526→        slug: form.slug.trim(), // server will slugify + de-dupe if empty
   527→        category: form.category.trim() || null,
   528→        subcategory: form.subcategory.trim() || null,
   529→        difficulty: form.difficulty || null,
   530→        excerpt: form.excerpt,
   531→        shortAnswer: form.shortAnswer,
   532→        content: form.content,
   533→        coverImageUrl: form.coverImageUrl,
   534→        authorName: form.authorName,
   535→        bibleRefs: form.bibleRefs, // CSV → server coerces to JSON array
   536→        status: form.status, // "draft" | "published"
   537→        featured: form.featured,
   538→        seoTitle: form.seoTitle,
   539→        seoDescription: form.seoDescription,
   540→        canonicalUrl: form.canonicalUrl,
   541→      };
   542→      const url = articleId ? `/api/admin/articles/${articleId}` : "/api/admin/articles";
   543→      const method = articleId ? "PATCH" : "POST";
   544→      const res = await fetch(url, {
   545→        method,
   546→        headers: { "Content-Type": "application/json" },
   547→        body: JSON.stringify(payload),
   548→      });
   549→      if (!res.ok) {
   550→        const data = await res.json().catch(() => ({}));
   551→        throw new Error(data?.error || `HTTP ${res.status}`);
   552→      }
   553→      toast.success(articleId ? "Article updated" : "Article created");
   554→      onSaved();
   555→    } catch (e: any) {
   556→      toast.error(e?.message || "Failed to save article");
   557→    } finally {
   558→      setSaving(false);
   559→    }
   560→  };
   561→
   562→  const remove = async () => {
   563→    if (!articleId) return;
   564→    setDeleting(true);
   565→    try {
   566→      const res = await fetch(`/api/admin/articles/${articleId}`, { method: "DELETE" });
   567→      if (!res.ok) {
   568→        const data = await res.json().catch(() => ({}));
   569→        throw new Error(data?.error || `HTTP ${res.status}`);
   570→      }
   571→      toast.success("Article deleted");
   572→      onDeleted();
   573→    } catch (e: any) {
   574→      toast.error(e?.message || "Failed to delete article");
   575→    } finally {
   576→      setDeleting(false);
   577→    }
   578→  };
   579→
   580→  if (loading) {
   581→    return (
   582→      <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/60 backdrop-blur-sm">
   583→        <div className="w-8 h-8 rounded-full border-2 border-transparent border-t-[#7C3AED] animate-spin" />
   584→      </div>
   585→    );
   586→  }
   587→
   588→  return (
   589→    <div
   590→      className="fixed inset-0 z-50 flex items-end sm:items-center justify-center p-0 sm:p-4 bg-black/60 backdrop-blur-sm"
   591→      onClick={onClose}

   717→          >

   719→              value={form.content}
   720→              onChange={(e) => set("content", e.target.value)}
   721→              className="neo-input text-sm h-44 resize-y font-mono"

   795→              checked={form.status === "published"}

   797→              label="Published"
   798→              hint={form.status === "published" ? "Visible on /apologetics" : "Hidden (draft)"}

   800→              color="#22C55E"
   801→            />
   802→            <Toggle
   803→              checked={form.featured}
   804→              onChange={(v) => set("featured", v)}
   805→              label="Featured"

   847→          <div className="flex flex-wrap items-center gap-2 pt-3 border-t border-white/[0.06]">
   848→            {articleId && (
   849→              <div className="mr-auto flex items-center gap-2">
   850→                {currentStatus === "published" && currentSlug && (
   851→                  <a
   852→                    href={`/apologetics/${currentSlug}`}
   853→                    target="_blank"
   854→                    rel="noopener noreferrer"
   855→                    className="text-[#38BDF8] hover:bg-[#38BDF8]/10 text-[11px] font-bold flex items-center gap-1.5 px-2 py-1 rounded-lg transition-all"

   862→                  <div className="flex items-center gap-1.5">

   864→                    <span className="text-[11px] text-[#94A3B8]">Delete permanently?</span>

   866→                      onClick={remove}
   867→                      disabled={deleting}

   869→                    >
   870→                      {deleting ? <Loader2 size={11} className="animate-spin" /> : "Yes, delete"}
   871→                    </button>
   872→                    <button
   873→                      onClick={() => setConfirmDelete(false)}
   874→                      disabled={deleting}
   875→                      className="text-[#94A3B8] hover:text-white text-[11px] font-bold px-2 py-1"
   876→                    >
   877→                      Cancel
   878→                    </button>
   879→                  </div>
   880→                ) : (
   881→                  <button
   882→                    onClick={() => setConfirmDelete(true)}
   883→                    disabled={saving || deleting}
   884→                    className="text-[#EF4444] hover:bg-[#EF4444]/10 text-[11px] font-bold flex items-center gap-1.5 disabled:opacity-50"
   885→                  >
   886→                    <Trash2 size={12} /> Delete
   887→                  </button>
   888→                )}
   889→              </div>
   890→            )}
   891→            <button
   892→              onClick={onClose}
   893→              disabled={saving || deleting}
   894→              className="px-4 py-2.5 rounded-xl bg-white/[0.04] text-[#94A3B8] hover:text-white border border-white/[0.06] text-sm font-semibold disabled:opacity-50"
   895→            >
   896→              Cancel
   897→            </button>
   898→            <button
   899→              onClick={save}
   900→              disabled={saving || deleting}
   901→              className="flex-1 sm:flex-none px-6 py-2.5 rounded-xl bg-[#7C3AED] hover:bg-[#6D28D9] text-white font-extrabold text-sm flex items-center justify-center gap-1.5 disabled:opacity-50 transition-all hover:-translate-y-px"
   902→            >
   903→              {saving ? (
   904→                <>
   905→                  <Loader2 size={14} className="animate-spin" /> Saving…
   906→                </>
   907→              ) : (
   908→                <>
   909→                  <Save size={14} /> {articleId ? "Save Changes" : "Create Article"}
   910→                </>
   911→              )}
   912→            </button>
   913→          </div>
   914→        </div>
   915→      </div>
   916→    </div>
   917→  );
   918→}

   922→function Field({
   923→  label,
   924→  required,
   925→  hint,
   926→  children,
   927→}: {
   928→  label: string;
   929→  required?: boolean;
   930→  hint?: string;
   931→  children: React.ReactNode;
   932→}) {
   933→  return (
   934→    <div>
   935→      <label className="block text-[11px] font-semibold uppercase tracking-wider text-[#94A3B8] mb-1.5">
   936→        {label} {required && <span className="text-[#F39B9B]">*</span>}
   937→      </label>
   938→      {children}
   939→      {hint && <p className="mt-1 text-[10px] text-[#64748B] leading-relaxed">{hint}</p>}
   940→    </div>
   941→  );
   942→}

   947→  label,
   948→  hint,
   949→  icon,

   962→      onClick={() => onChange(!checked)}

   964→        checked ? "border-white/[0.15]" : "border-white/[0.06] hover:border-white/[0.1]"

   972→        {icon}
   973→      </span>
   974→      <div className="min-w-0 flex-1">
   975→        <p className="text-xs font-bold text-white">{label}</p>
   976→        <p className="text-[10px] text-[#94A3B8]">{hint}</p>

   986→            <path
   987→              d="M2 5L4 7L8 3"

   989→              strokeWidth="2"

   992→            />
   993→          </svg>

   995→      </span>
   996→    </button>
