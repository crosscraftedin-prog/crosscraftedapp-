   372→function MembersTab() {
   373→  const [members, setMembers] = useState<any[]>([]);
   374→  const [stats, setStats] = useState<any>(null);
   375→  const [loading, setLoading] = useState(true);
   376→  const [search, setSearch] = useState("");
   377→  const [filterState, setFilterState] = useState("");
   378→  const [filterStatus, setFilterStatus] = useState("");
   379→  const [filterVerified, setFilterVerified] = useState("");
   380→  const [total, setTotal] = useState(0);
   381→  const [page, setPage] = useState(0);
   382→  const limit = 20;
   383→  const [selectedMember, setSelectedMember] = useState<any>(null);
   384→  const [memberDetail, setMemberDetail] = useState<any>(null);
   385→  const [detailLoading, setDetailLoading] = useState(false);
   386→  const [saving, setSaving] = useState(false);
   387→
   388→  const loadMembers = useCallback(async () => {
   389→    setLoading(true);
   390→    try {
   391→      const params = new URLSearchParams({ limit: String(limit), offset: String(page * limit) });
   392→      if (search) params.set("search", search);
   393→      if (filterState) params.set("state", filterState);
   394→      if (filterStatus) params.set("profileCompleted", filterStatus);
   395→      const res = await fetch(`/api/admin/members?${params}`);
   396→      const data = await res.json();
   397→      setMembers(data.members || []);
   398→      setTotal(data.total || 0);
   399→    } catch { setMembers([]); }
   400→    finally { setLoading(false); }
   401→  }, [search, filterState, filterStatus, page]);
   402→
   403→  const loadStats = useCallback(async () => {
   404→    try {
   405→      const res = await fetch("/api/admin/members?stats=true");
   406→      const data = await res.json();
   407→      setStats(data);
   408→    } catch {}
   409→  }, []);
   410→
   411→  useEffect(() => { loadMembers(); loadStats(); }, [loadMembers, loadStats]);
   412→
   413→  const openMember = async (id: string) => {
   414→    setSelectedMember(id);
   415→    setDetailLoading(true);
   416→    try {
   417→      const res = await fetch(`/api/admin/members/${id}`);
   418→      const data = await res.json();
   419→      setMemberDetail(data.member);
   420→    } catch {}
   421→    finally { setDetailLoading(false); }
   422→  };
   423→
   424→  const updateMember = async (updates: any) => {
   425→    if (!memberDetail) return;
   426→    setSaving(true);
   427→    try {
   428→      const res = await fetch(`/api/admin/members/${memberDetail.id}`, {
   429→        method: "PATCH",
   430→        headers: { "Content-Type": "application/json" },
   431→        body: JSON.stringify(updates),
   432→      });
   433→      const data = await res.json();
   434→      if (!res.ok) throw new Error(data.error);
   435→      setMemberDetail(data.member);
   436→      toast.success("Member updated");
   437→      loadMembers();
   438→    } catch (e: any) {
   439→      toast.error(e.message || "Failed to update");
   440→    } finally { setSaving(false); }
   441→  };
   442→
   443→  const togglePermission = (perm: string) => {
   444→    if (!memberDetail) return;
   445→    const current: string[] = JSON.parse(memberDetail.permissions || "[]");
   446→    const next = current.includes(perm) ? current.filter(p => p !== perm) : [...current, perm];
   447→    updateMember({ permissions: next });
   448→  };
   449→
   450→  return (
   451→    <div className="space-y-4">
   452→      <PreviewModeBanner section="Members" />
   453→
   454→      {/* Stats */}
   455→      {stats && (
   456→        <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   457→          <div className="bg-[#1C1929] border border-white/[0.06] rounded-xl p-3">
   458→            <p className="text-xl font-extrabold text-white">{stats.totalMembers}</p>
   459→            <p className="text-[9px] text-[#94A3B8] uppercase">Total Members</p>
