   293→function ChurchesTab() {
   294→  const [churches, setChurches] = useState<Church[]>(CHURCHES);
   295→
   296→  const approve = (id: string) => {
   297→    setChurches((cs) => cs.map((c) => (c.id === id ? { ...c, status: "verified" } : c)));
   298→    toast.success("Church approved!", { description: "It's now visible in the public directory." });
   299→  };
   300→
   301→  const remove = (id: string) => {
   302→    setChurches((cs) => cs.filter((c) => c.id !== id));
   303→    toast("Church removed");
   304→  };
   305→
   306→  const pending = churches.filter((c) => c.status === "pending");
   307→  const verified = churches.filter((c) => c.status === "verified");
   308→
   309→  return (
   310→    <div className="space-y-4">
   311→      <AdminSectionHeader title="Pending Approval" count={pending.length} color="#F59E0B" />
   312→      {pending.length === 0 ? (
   313→        <EmptyState text="No pending churches. All caught up!" />
   314→      ) : (
   315→        <div className="space-y-2">
   316→          {pending.map((c) => (
   317→            <AdminCard
   318→              key={c.id}
   319→              title={c.name}
   320→              subtitle={`${c.city}, ${c.state} · ${c.denomination}`}
   321→              description={c.description}
   322→              image={c.cover_image}
   323→              actions={
   324→                <>
   325→                  <button
   326→                    onClick={() => approve(c.id)}
   327→                    className="flex items-center gap-1 px-3 py-1.5 rounded-lg bg-[#22C55E]/15 border border-[#22C55E]/30 text-[#22C55E] text-xs font-bold hover:bg-[#22C55E]/25 transition-all"
   328→                  >
   329→                    <Check size={12} /> Approve
   330→                  </button>
   331→                  <button
   332→                    onClick={() => remove(c.id)}
   333→                    className="flex items-center gap-1 px-3 py-1.5 rounded-lg bg-[#EF4444]/15 border border-[#EF4444]/30 text-[#EF4444] text-xs font-bold hover:bg-[#EF4444]/25 transition-all"
   334→                  >
   335→                    <Trash2 size={12} /> Reject
   336→                  </button>
   337→                </>
   338→              }
   339→            />
   340→          ))}
   341→        </div>
   342→      )}
   343→
   344→      <AdminSectionHeader title="Verified Churches" count={verified.length} color="#22C55E" />
   345→      <div className="space-y-2">
   346→        {verified.map((c) => (
   347→          <AdminCard
   348→            key={c.id}
   349→            title={c.name}
   350→            subtitle={`${c.city}, ${c.state} · ${c.followers_count} followers`}
   351→            description={c.description}
   352→            image={c.cover_image}
   353→            actions={
   354→              <button
   355→                onClick={() => remove(c.id)}
   356→                className="flex items-center gap-1 px-3 py-1.5 rounded-lg bg-white/[0.04] border border-white/[0.06] text-[#94A3B8] hover:text-[#EF4444] text-xs font-bold transition-all"
   357→              >
   358→                <Trash2 size={12} /> Remove
   359→              </button>
   360→            }
   361→          />
   362→        ))}
   363→      </div>
   364→    </div>
   365→  );
   366→}

   410→function MarketplaceTab() {
   411→  const [products, setProducts] = useState<Product[]>(PRODUCTS);
   412→
   413→  const remove = (id: string) => {
   414→    setProducts((ps) => ps.filter((p) => p.id !== id));

   416→  };
   417→
   418→  const toggleStock = (id: string) => {
   419→    setProducts((ps) =>
   420→      ps.map((p) => (p.id === id ? { ...p, in_stock: !p.in_stock } : p))
   421→    );
   422→    toast("Stock updated");
   423→  };
   424→
   425→  return (
   426→    <div className="space-y-3">
   427→      <AdminSectionHeader title="All Products" count={products.length} color="#9786E3" />
   428→      {products.map((p) => (
   429→        <AdminCard
   430→          key={p.id}
   431→          title={p.name}
   432→          subtitle={`₹${p.price} · ${p.vendor} · ${p.city}`}
   433→          description={p.description}
   434→          image={p.cover_image}
   435→          badge={p.in_stock ? "In Stock" : "Out"}
   436→          badgeColor={p.in_stock ? "#22C55E" : "#EF4444"}
   437→          actions={
   438→            <>
   439→              <button
   440→                onClick={() => toggleStock(p.id)}
   441→                className="flex items-center gap-1 px-3 py-1.5 rounded-lg bg-white/[0.04] border border-white/[0.06] text-[#94A3B8] hover:text-white text-xs font-bold transition-all"
   442→              >
   443→                {p.in_stock ? "Mark Out" : "Mark In"}
   444→              </button>
   445→              <button
   446→                onClick={() => remove(p.id)}
   447→                className="flex items-center gap-1 px-3 py-1.5 rounded-lg bg-[#EF4444]/15 border border-[#EF4444]/30 text-[#EF4444] text-xs font-bold hover:bg-[#EF4444]/25 transition-all"

   450→              </button>
   451→            </>
   452→          }

   454→      ))}

   457→}

   500→function ApologeticsTab() {
   501→  const [questions, setQuestions] = useState<ApologeticsQuestion[]>(APOLOGETICS_QUESTIONS);
   502→  const [answerText, setAnswerText] = useState<Record<string, string>>({});
   503→
   504→  const postAnswer = (questionId: string) => {
   505→    const text = (answerText[questionId] || "").trim();
   506→    if (!text) {
   507→      toast.error("Please write your answer first");
   508→      return;
   509→    }
   510→    setQuestions((qs) =>
   511→      qs.map((q) =>
   512→        q.id === questionId
   513→          ? {
   514→              ...q,
   515→              status: "answered",
   516→              answers: [
   517→                ...q.answers,
   518→                {
   519→                  id: `admin_${Date.now()}`,
   520→                  author: "Admin (Pastor)",
   521→                  authorRole: "Pastor",
   522→                  body: text,
   523→                  date: new Date().toISOString().split("T")[0],
   524→                  is_accepted: true,
   525→                  likes: 0,
   526→                },
   527→              ],
   528→            }
   529→          : q
   530→      )
   531→    );
   532→    setAnswerText((prev) => ({ ...prev, [questionId]: "" }));
   533→    toast.success("Official answer posted!", { description: "Marked as accepted answer." });
   534→  };
   535→
   536→  const closeQuestion = (id: string) => {
   537→    setQuestions((qs) => qs.map((q) => (q.id === id ? { ...q, status: "closed" } : q)));
   538→    toast("Question closed");
   539→  };
   540→
   541→  const remove = (id: string) => {
   542→    setQuestions((qs) => qs.filter((q) => q.id !== id));
   543→    toast("Question removed");
   544→  };
   545→
   546→  return (
   547→    <div className="space-y-3">
   548→      <AdminSectionHeader title="Apologetics Q&A" count={questions.length} color="#38BDF8" />
   549→      {questions.map((q) => (
   550→        <div key={q.id} className="bg-[#1C1929] border border-white/[0.06] rounded-2xl p-4">
   551→          <div className="flex items-center gap-2 mb-2">
   552→            <span
   553→              className={`px-2 py-0.5 rounded-full text-[10px] font-bold uppercase tracking-wider ${
   554→                q.status === "answered"
   555→                  ? "bg-[#22C55E]/15 text-[#22C55E]"
   556→                  : q.status === "closed"
   557→                  ? "bg-white/[0.06] text-[#94A3B8]"
   558→                  : "bg-[#F59E0B]/15 text-[#F59E0B]"
   559→              }`}
   560→            >
   561→              {q.status}
   562→            </span>
   563→            <span className="text-[10px] text-[#94A3B8]">{q.author}</span>
   564→            <span className="text-[10px] text-[#64748B]">· {q.answers.length} answers</span>
   565→          </div>
   566→          <h3 className="text-sm font-bold text-white mb-1">{q.title}</h3>
   567→          <p className="text-[12px] text-[#A09DB1] line-clamp-2 mb-3">{q.body}</p>
   568→
   569→          {/* Answer input */}
   570→          <div className="bg-white/[0.02] border border-white/[0.06] rounded-xl p-2 mb-2">
   571→            <textarea
   572→              value={answerText[q.id] || ""}
   573→              onChange={(e) => setAnswerText((prev) => ({ ...prev, [q.id]: e.target.value }))}
   574→              placeholder="Post an official answer as Pastor/Admin..."
   575→              className="w-full bg-transparent text-xs text-white outline-none resize-none h-16 placeholder:text-[#64748B]"
   576→            />
   577→            <div className="flex gap-2 mt-1">
   578→              <button
   579→                onClick={() => postAnswer(q.id)}
   580→                className="flex items-center gap-1 px-3 py-1.5 rounded-lg bg-[#38BDF8]/15 border border-[#38BDF8]/30 text-[#38BDF8] text-xs font-bold hover:bg-[#38BDF8]/25 transition-all"
   581→              >
   582→                <Send size={11} /> Post Official Answer
   583→              </button>
   584→              {q.status !== "closed" && (
   585→                <button
   586→                  onClick={() => closeQuestion(q.id)}
   587→                  className="px-3 py-1.5 rounded-lg bg-white/[0.04] border border-white/[0.06] text-[#94A3B8] hover:text-white text-xs font-bold transition-all"
   588→                >
   589→                  Close
   590→                </button>
   591→              )}
   592→              <button
   593→                onClick={() => remove(q.id)}
   594→                className="flex items-center gap-1 px-3 py-1.5 rounded-lg bg-[#EF4444]/15 border border-[#EF4444]/30 text-[#EF4444] text-xs font-bold hover:bg-[#EF4444]/25 transition-all ml-auto"
   595→              >
   596→                <Trash2 size={11} /> Remove
   597→              </button>
   598→            </div>
   599→          </div>
   600→        </div>
   601→      ))}
   602→    </div>
   603→  );
   604→}

   759→// ─── GIFTS MANAGEMENT ────────────────────────────────────────────────────
   760→
   761→function GiftsManagement() {
   762→  const [gifts, setGifts] = useState<any[]>([]);
   763→  const [loading, setLoading] = useState(true);
   764→  const [showAddForm, setShowAddForm] = useState(false);
   765→  const [editingId, setEditingId] = useState<string | null>(null);
   766→  const [form, setForm] = useState<{
   767→    title: string;
   768→    description: string;
   769→    images: string[];
   770→    points_required: string;
   771→    tier: "bronze" | "silver" | "gold" | "platinum";
   772→    stock: string;
   773→    variations: { name: string; options: string[] }[];
   774→    attributes: { label: string; value: string }[];
   775→  }>({
   776→    title: "",
   777→    description: "",
   778→    images: [],
   779→    points_required: "",
   780→    tier: "bronze",
   781→    stock: "10",
   782→    variations: [],
   783→    attributes: [],
   784→  });
   785→
   786→  const fetchGifts = useCallback(async () => {
   787→    try {
   788→      const res = await fetch("/api/admin/gifts");
   789→      const data = await res.json();
   790→      const mapped = (data.gifts || []).map((g: any) => ({
   791→        id: g.id,
   792→        title: g.title,
   793→        description: g.description,
   794→        image_url: g.imageUrl,
   795→        points_required: g.pointsRequired,
   796→        tier: g.tier,
   797→        stock: g.stock,
   798→        isAdmin: g.isAdmin,
   799→        variations: Array.isArray(g.variations) ? g.variations : [],
   800→        attributes: Array.isArray(g.attributes) ? g.attributes : [],
   801→      }));
   802→      setGifts(mapped);
   803→    } catch {
   804→      toast.error("Failed to load gifts");
   805→    } finally {
   806→      setLoading(false);
   807→    }
   808→  }, []);
   809→
   810→  useEffect(() => {
   811→    fetchGifts();
   812→  }, [fetchGifts]);
   813→
   814→  const resetForm = () => {
   815→    setForm({
   816→      title: "",
   817→      description: "",
   818→      images: [],
   819→      points_required: "",
   820→      tier: "bronze",
   821→      stock: "10",
   822→      variations: [],
   823→      attributes: [],
   824→    });
   825→    setEditingId(null);
   826→    setShowAddForm(false);
   827→  };
   828→
   829→  const handleSubmit = async (e: React.FormEvent) => {
   830→    e.preventDefault();
   831→    if (!form.title.trim() || !form.points_required || form.images.length === 0) {
   832→      toast.error("Please fill in title, points, and add at least 1 image");
   833→      return;
   834→    }
   835→    if (form.variations.some((v) => v.name.trim() && v.options.length === 0)) {
   836→      toast.error("Add at least 1 option for each variation name");
   837→      return;
   838→    }
   839→
   840→    const giftData = {
   841→      title: form.title.trim(),
   842→      description: form.description.trim(),
   843→      imageUrl: form.images[0],
   844→      pointsRequired: Number(form.points_required),
   845→      tier: form.tier,
   846→      stock: Number(form.stock) || 0,
   847→      variations: form.variations.filter((v) => v.name.trim() && v.options.length > 0),
   848→      attributes: form.attributes.filter((a) => a.label.trim() && a.value.trim()),
   849→    };
   850→
   851→    try {
   852→      if (editingId) {
   853→        const res = await fetch("/api/admin/gifts", {
   854→          method: "PUT",
   855→          headers: { "Content-Type": "application/json" },
   856→          body: JSON.stringify({ giftId: editingId, ...giftData }),
   857→        });
   858→        if (!res.ok) throw new Error("Failed to update");
   859→        toast.success("Gift updated!", { description: "Changes are live in Trivia > Rewards." });
   860→      } else {
   861→        const res = await fetch("/api/admin/gifts", {
   862→          method: "POST",
   863→          headers: { "Content-Type": "application/json" },
   864→          body: JSON.stringify(giftData),
   865→        });
   866→        if (!res.ok) throw new Error("Failed to add");
   867→        toast.success("Gift added!", {
   868→          description: "It's now visible in Trivia > Rewards for players to redeem.",
   869→        });
   870→      }
   871→      fetchGifts();
   872→      resetForm();
   873→    } catch {
   874→      toast.error("Failed to save gift");
   875→    }
   876→  };
   877→
   878→  const handleEdit = (gift: any) => {
   879→    if (!gift.isAdmin) {
   880→      toast("Default gifts can't be edited", {
   881→        description: "Only gifts you've added from the admin panel can be modified.",
   882→      });
   883→      return;
   884→    }
   885→    setEditingId(gift.id);
   886→    setForm({
   887→      title: gift.title,
   888→      description: gift.description,
   889→      images: [gift.image_url],
   890→      points_required: String(gift.points_required),
   891→      tier: gift.tier,
   892→      stock: String(gift.stock),
   893→      variations: Array.isArray(gift.variations) && gift.variations.length > 0
   894→        ? gift.variations.map((v: any) => ({
   895→            name: v.name || "",
   896→            options: Array.isArray(v.options) ? v.options : [],
   897→          }))
   898→        : [],
   899→      attributes: Array.isArray(gift.attributes) && gift.attributes.length > 0
   900→        ? gift.attributes.map((a: any) => ({
   901→            label: a.label || "",
   902→            value: a.value || "",
   903→          }))
   904→        : [],
   905→    });
   906→    setShowAddForm(true);
   907→    setTimeout(() => {
   908→      document.getElementById("gift-form")?.scrollIntoView({ behavior: "smooth", block: "center" });
   909→    }, 100);
   910→  };
   911→
   912→  const handleRemove = async (id: string) => {
   913→    const gift = gifts.find((g) => g.id === id);
   914→    if (!gift?.isAdmin) {
   915→      toast("Default gifts can't be removed", {
   916→        description: "Only gifts you've added from the admin panel can be deleted.",
   917→      });
   918→      return;
   919→    }
   920→    try {
   921→      const res = await fetch("/api/admin/gifts", {
   922→        method: "DELETE",
   923→        headers: { "Content-Type": "application/json" },
   924→        body: JSON.stringify({ giftId: id }),
   925→      });
   926→      if (!res.ok) throw new Error("Failed to delete");
   927→      toast("Gift removed", { description: "Players can no longer redeem this gift." });
   928→      fetchGifts();
   929→    } catch {
   930→      toast.error("Failed to remove gift");
   931→    }
   932→  };
   933→
   934→  const handleStockChange = async (id: string, delta: number) => {
   935→    const gift = gifts.find((g) => g.id === id);
   936→    if (!gift?.isAdmin) return;
   937→    const newStock = Math.max(0, gift.stock + delta);
   938→    try {
   939→      await fetch("/api/admin/gifts", {
   940→        method: "PUT",
   941→        headers: { "Content-Type": "application/json" },
   942→        body: JSON.stringify({ giftId: id, stock: newStock }),
   943→      });
   944→      fetchGifts();
   945→    } catch {
   946→      toast.error("Failed to update stock");
   947→    }
   948→  };
   949→
   950→  const tierColors: Record<string, string> = {
   951→    bronze: "#CD7F32",
   952→    silver: "#C0C0C0",
   953→    gold: "#FFD700",
   954→    platinum: "#E5E4E2",
   955→  };
   956→
   957→  const adminCount = gifts.filter((g) => g.isAdmin).length;
   958→
   959→  if (loading) {
   960→    return (
   961→      <div className="flex items-center justify-center py-8">
   962→        <div className="w-8 h-8 rounded-full border-2 border-transparent border-t-[#F59E0B] animate-spin" />
   963→      </div>
   964→    );
   965→  }
   966→
   967→  return (
   968→    <div className="space-y-3">
   969→      <div className="flex items-center justify-between">
   970→        <AdminSectionHeader title="Reward Gifts" count={gifts.length} color="#F59E0B" />
   971→        <button
   972→          onClick={() => {
   973→            if (showAddForm) {
   974→              resetForm();
   975→            } else {
   976→              setShowAddForm(true);
   977→            }
   978→          }}
   979→          className="flex items-center gap-1.5 px-3 py-2 rounded-xl bg-[#F59E0B] hover:bg-[#E59E0B] text-slate-950 text-xs font-bold transition-all"
   980→        >
   981→          {showAddForm ? <X size={12} /> : <Plus size={12} />}
   982→          {showAddForm ? "Cancel" : "Add Gift"}
   983→        </button>
   984→      </div>
   985→
   986→      {adminCount > 0 && (
   987→        <p className="text-[10px] text-[#64748B] px-1">
   988→          {adminCount} admin-added · {gifts.length - adminCount} default gifts
   989→        </p>
   990→      )}
   991→
   992→      <AnimatePresence>
   993→        {showAddForm && (
   994→          <motion.form
   995→            id="gift-form"
   996→            initial={{ opacity: 0, height: 0 }}
   997→            animate={{ opacity: 1, height: "auto" }}
   998→            exit={{ opacity: 0, height: 0 }}
   999→            onSubmit={handleSubmit}
  1000→            className="bg-[#1C1929] border border-[#F59E0B]/20 rounded-2xl p-4 space-y-3 overflow-hidden"
  1001→          >
  1002→            <div className="flex items-center gap-2 mb-2">
  1003→              <Gift size={14} className="text-[#F59E0B]" />
  1004→              <p className="text-[11px] font-bold uppercase tracking-wider text-[#F59E0B]">
  1005→                {editingId ? "Edit Gift" : "Add New Reward Gift"}
  1006→              </p>
  1007→            </div>
  1008→
  1009→            <div>
  1010→              <label className="block text-[11px] font-semibold uppercase tracking-wider text-[#94A3B8] mb-1.5">
  1011→                Gift Title *
  1012→              </label>
  1013→              <input
  1014→                type="text"
  1015→                value={form.title}
  1016→                onChange={(e) => setForm({ ...form, title: e.target.value })}

  1018→                placeholder="e.g. Believ Hoodie"
  1019→                required

  1021→            </div>

  1023→            <div>

  1025→                Description
