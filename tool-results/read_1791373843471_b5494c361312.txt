   158→// ─── ANTI-FARMING: Question Attempts ───────────────────────────────────────
   159→// THE KEY TABLE: records that a user has already earned points for a question.
   160→// @@unique([userId, questionId]) means a user can only have ONE attempt
   161→// record per question — ever. This is enforced at the DATABASE level,
   162→// so even race conditions (two simultaneous requests) cannot create duplicates.
   163→
   164→model TriviaQuestionAttempt {
   165→  id              String   @id @default(cuid())
   166→  userId          String
   167→  questionId      String   // references TriviaQuestion.questionId (stable ID)
   168→  quizSessionId   String?
   169→  pointsAwarded   Int      @default(0) // 0 if practice/already-scored, 10-60 if new
   170→  answeredCorrectly Boolean
   171→  answeredAt      DateTime @default(now())
   172→
   173→  user            User     @relation(fields: [userId], references: [id], onDelete: Cascade)
   174→  question        TriviaQuestion @relation(fields: [questionId], references: [questionId], onDelete: Cascade)
   175→  quizSession     TriviaQuizSession? @relation(fields: [quizSessionId], references: [id])
   176→
   177→  @@unique([userId, questionId]) // ← ANTI-FARMING: one score per question per user, EVER
   178→  @@index([userId])
   179→  @@index([questionId])
   180→}

   472→// ─── TRIVIA COMPETITIONS ───────────────────────────────────────────────────
   473→// A Competition is a time-boxed Bible Trivia contest with its own
   474→// server-authoritative score (separate from lifetime Faith Points).
   475→// Examples: Daily Challenge, Weekly Bible Challenge, Monthly Championship.
   476→//
   477→// KEY DESIGN DECISIONS:
   478→// - Competition Score is SEPARATE from lifetime FP. The weekly leaderboard
   479→//   ranks by competition score, NOT by lifetime FP.
   480→// - attemptLimit is enforced SERVER-SIDE (not in localStorage).
   481→// - Questions are server-selected — the client never sees correct answers
   482→//   before submitting.
   483→// - Winners are calculated server-side when endAt passes. Immutable records.
   484→
   485→model TriviaCompetition {
   486→  id              String   @id @default(cuid())
   487→  title           String
   488→  description     String?
   489→  imageUrl        String?
   490→  type            String   // "daily" | "weekly" | "monthly" | "custom"
   491→  category        String   // "full_bible" | "new_testament" | "old_testament" | "apologetics" | "mixed"
   492→  difficulty      String   // "beginners" | "intermediate" | "skilled" | "expert" | "mixed"
   493→  questionCount   Int      @default(10)
   494→  attemptLimit    Int      @default(1)
   495→  winnerCount     Int      @default(3)
   496→  rules           String?  // free-text rules displayed to users before participating
   497→  status          String   @default("draft") // "draft" | "scheduled" | "live" | "ended" | "cancelled"
   498→  startAt         DateTime
   499→  endAt           DateTime
   500→  claimDeadlineDays Int    @default(7) // days after competition ends to claim prize
   501→  prizeId         String?  // references TriviaPrize.id (optional — some competitions may not have prizes)
   502→  createdById     String?  // admin userId who created it
   503→  createdAt       DateTime @default(now())
   504→  updatedAt       DateTime @updatedAt
   505→
   506→  prize           TriviaPrize?  @relation(fields: [prizeId], references: [id])
   507→  attempts        TriviaCompetitionAttempt[]
   508→  winners         TriviaWinner[]
   509→
   510→  @@index([status])
   511→  @@index([type])
   512→  @@index([endAt])
   513→}

   515→// ─── COMPETITION ATTEMPTS ──────────────────────────────────────────────────
